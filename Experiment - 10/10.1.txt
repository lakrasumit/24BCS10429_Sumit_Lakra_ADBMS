EXPERIMENT - 10
FIND SECOND AND THIRD HIGHEST SALARIES:
CODE:
CREATE TABLE Employee (
    emp_id INT PRIMARY KEY,
    emp_name VARCHAR(30),
    salary INT
);

INSERT INTO Employee VALUES
(1, 'Aman', 50000),
(2, 'Riya', 70000),
(3, 'Karan', 60000),
(4, 'Neha', 90000),
(5, 'Raj', 80000),
(6, 'Simran', 70000);

SELECT MAX(salary) AS second_highest
FROM Employee
WHERE salary < (
    SELECT MAX(salary)
    FROM Employee
);
SELECT MAX(salary) AS third_highest
FROM Employee
WHERE salary < (
    SELECT MAX(salary)
    FROM Employee
    WHERE salary < (
        SELECT MAX(salary)
        FROM Employee
    )
);
