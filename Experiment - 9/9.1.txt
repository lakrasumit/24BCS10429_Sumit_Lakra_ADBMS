9.1 :Implement a Row-Level BEFORE UPDATE Trigger on the Salary_Hike table that restricts a salary
increase to no more than 15% of the :OLD.salary value; if the increase exceeds this limit, the trigger must raise a custom User-Defined Exception 
with a specific message

Code :
CREATE TABLE Salary_Hike (
    emp_id INT PRIMARY KEY,
    emp_name VARCHAR(100),
    salary NUMERIC(10,2)
);

INSERT INTO Salary_Hike VALUES
(101, 'Amit', 50000),
(102, 'Rahul', 60000),
(103, 'Priya', 55000);

CREATE OR REPLACE FUNCTION CHECK_SALARY_HIKE()
RETURNS TRIGGER
AS
$$
BEGIN
    IF NEW.salary > OLD.salary * 1.15 THEN
        RAISE EXCEPTION
        'Salary hike cannot exceed 15 percent of old salary';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE PLPGSQL;

CREATE OR REPLACE TRIGGER TRG_SALARY_HIKE
BEFORE UPDATE ON Salary_Hike
FOR EACH ROW
EXECUTE FUNCTION CHECK_SALARY_HIKE();

-- Testing
--WORKS FINE
UPDATE Salary_Hike
SET salary = 57000
WHERE emp_id = 101;

--RAISES EXCEPTION
UPDATE Salary_Hike
SET salary = 72000
WHERE emp_id = 102;
